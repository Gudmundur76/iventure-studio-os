S6-BDA-01 Stripe Tax Automation
src/billing/stripe-tax.tsimport Stripe from 'stripe';
import { db } from '../config/database';
import { logger } from '../observability/logger';

const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!, { apiVersion: '2025-10-01' });

// EU country → Stripe tax rate ID mapping (pre-created in Stripe dashboard)
const EU_TAX_RATES: Record<string, string> = {
  IS: process.env.STRIPE_TAX_IS!, // Iceland VAT 24%
  NO: process.env.STRIPE_TAX_NO!, // Norway MVA 25%
  SE: process.env.STRIPE_TAX_SE!, // Sweden Moms 25%
  DK: process.env.STRIPE_TAX_DK!, // Denmark Moms 25%
  FI: process.env.STRIPE_TAX_FI!, // Finland ALV 24%
  DE: process.env.STRIPE_TAX_DE!, // Germany MwSt 19%
  NL: process.env.STRIPE_TAX_NL!, // Netherlands BTW 21%
  GB: process.env.STRIPE_TAX_GB!, // UK VAT 20%
};

export async function applyTaxToCheckout(
  session_params: Stripe.Checkout.SessionCreateParams,
  company_id: string
): Promise<Stripe.Checkout.SessionCreateParams> {
  const company = await db.query(
    'SELECT country, vat_number, is_b2b FROM companies WHERE id = $1',
    [company_id]
  );
  const { country, vat_number, is_b2b } = company.rows[0] ?? {};

  // B2B with valid VAT number → reverse charge (0% VAT)
  if (is_b2b && vat_number && await validateVatNumber(vat_number, country)) {
    logger.info({ company_id, vat_number }, 'B2B reverse charge applied');
    return {
      ...session_params,
      automatic_tax: { enabled: false },
      customer_update: { address: 'auto' },
    };
  }

  // Apply country-specific tax rate
  const taxRateId = EU_TAX_RATES[country];
  if (!taxRateId) {
    logger.warn({ company_id, country }, 'No tax rate for country, applying automatic tax');
    return { ...session_params, automatic_tax: { enabled: true } };
  }

  return {
    ...session_params,
    line_items: session_params.line_items?.map(item => ({
      ...item,
      tax_rates: [taxRateId],
    })),
  };
}

async function validateVatNumber(vat_number: string, country: string): Promise<boolean> {
  // VIES API validation (EU VAT validation service)
  try {
    const res = await fetch(
      `https://ec.europa.eu/taxation_customs/vies/rest-api/ms/${country}/vat/${vat_number.replace(/[^0-9A-Z]/g, '')}`
    );
    const data = await res.json();
    return data.isValid === true;
  } catch {
    logger.warn({ vat_number, country }, 'VIES validation failed, assuming valid');
    return true; // Fail open — do not block checkout
  }
}
S6-BDA-02 Zapier / Make.com Outbound Webhook
src/integrations/outbound-webhook.tsexport async function dispatchWebhook(event: WebhookPayload['event'], company_id: string, data: Record<string, unknown>): Promise<void> {
  // Get all active webhooks for this company
  const result = await db.query(
    `SELECT id, url, secret FROM company_webhooks WHERE company_id = $1 AND active = true AND $2 = ANY(events)`,
    [company_id, event]
  );

  if (result.rows.length === 0) return;

  const payload: WebhookPayload = {
    event,
    company_id,
    data,
    timestamp: new Date().toISOString(),
    version: '1.0',
  };
  const body = JSON.stringify(payload);

  await Promise.allSettled(result.rows.map(async (webhook) => {
    const sig = createHmac('sha256', webhook.secret).update(body).digest('hex');
    try {
      const res = await fetch(webhook.url, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-iVenture-Signature': `sha256=${sig}`,
          'X-iVenture-Event': event,
          'User-Agent': 'iVenture-Studio/1.0',
        },
        body,
        signal: AbortSignal.timeout(10_000),
      });
      await db.query(
        `UPDATE company_webhooks SET last_triggered_at = NOW(), last_status_code = $1 WHERE id = $2`,
        [res.status, webhook.id]
      );
      logger.info({ event, company_id, webhook_id: webhook.id, status: res.status }, 'Webhook dispatched');
    } catch (err) {
      logger.error({ event, company_id, webhook_id: webhook.id, err }, 'Webhook delivery failed');
    }
  }));
}
S6-BDA-03 Skill Result Caching (Redis TTL)
src/core/skill-cache.tsconst CACHE_TTL_SECONDS: Record<string, number> = {
  'competitive-intel':  3600,   // 1 hour — market data stays fresh
  'cashflow-forecaster': 1800,  // 30 min
  'translator':         86400,  // 24 hours — translations are stable
  'patent-research':    7200,   // 2 hours
  'crm-manager':        300,    // 5 min — customer data changes often
  'invoice-manager':    0,      // Never cache — always fresh invoice numbers
};

export async function getCachedResult(skill_id: string, input_hash: string): Promise<unknown | null> {
  const ttl = CACHE_TTL_SECONDS[skill_id];
  if (ttl === 0) return null;

  const key = `skill:cache:${skill_id}:${input_hash}`;
  const cached = await redis.get(key);
  if (cached) {
    logger.debug({ skill_id, input_hash }, 'Cache hit — skipping LLM call');
    return JSON.parse(cached);
  }
  return null;
}
