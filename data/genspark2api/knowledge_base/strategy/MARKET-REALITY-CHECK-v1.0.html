Section 7 — GDPRRouter Code Snippet (TypeScript)

// src/routing/gdpr-router.ts
// REPLACES: src/routing/pipl-router.ts (DELETED)

const EU_COUNTRIES = new Set([
  'IS','IE','GB','DE','FR','NL','BE','DK','SE','NO','FI',
  'AT','CH','PL','ES','IT','PT','CZ','HU','RO','GR','HR',
  'BG','SK','SI','EE','LV','LT','LU','MT','CY'
]);

export function routeRequest(countryCode: string, tld: string): string {
  // Hard block: China TLDs
  if (tld === '.cn') {
    throw new Error('403: China routing disabled per strategy directive 2026-03-18');
  }
  if (EU_COUNTRIES.has(countryCode)) return 'eu-central-1';
  if (['US','CA'].includes(countryCode)) return 'us-east-1';
  if (['AU','NZ'].includes(countryCode)) return 'ap-southeast-2';
  return 'eu-central-1'; // default: EU
}

Section 8 — Schema Migration SQL

-- Migration: Remove China-specific columns, add GDPR columns
-- File: postgres-migrate-v1.3.sql
-- Date: 2026-03-18
-- Author: BDA-v1 (redirected by strategy directive)

ALTER TABLE companies
  DROP COLUMN IF EXISTS icp_license_number,
  DROP COLUMN IF EXISTS pipl_consent_date,
  DROP COLUMN IF EXISTS csl_data_category,
  ADD COLUMN IF NOT EXISTS gdpr_consent_date TIMESTAMPTZ,
  ADD COLUMN IF NOT EXISTS data_residency_region VARCHAR(20)
    CHECK (data_residency_region IN ('eu-central-1','us-east-1','ap-southeast-2'))
    DEFAULT 'eu-central-1';

-- Index for region-based queries
CREATE INDEX IF NOT EXISTS idx_companies_region 
  ON companies(data_residency_region);

COMMENT ON COLUMN companies.data_residency_region 
  IS 'GDPR data residency: eu-central-1 (default), us-east-1, ap-southeast-2. CN routing disabled.';
