S4-BDA-01 Invoice Manager Skill

Full implementation of the invoice-manager skill including database storage and LLM generation.

src/skills/finance/invoice-manager.tsimport { z } from 'zod';
import { LLMClient } from '../../llm/llm-client';
import { CostTracker } from '../../core/cost-tracker';
import { db } from '../../config/database';
import { logger } from '../../observability/logger';

const InvoiceSchema = z.object({
  company_id: z.string().uuid(),
  client_name: z.string().min(1),
  items: z.array(z.object({
    description: z.string(),
    quantity: z.number().positive(),
    unit_price: z.number().positive(),
    vat_rate: z.number().min(0).max(1).default(0.25),
  })),
  currency: z.enum(['EUR', 'GBP', 'USD', 'NOK', 'SEK', 'DKK', 'ISK']).default('EUR'),
  due_days: z.number().default(30),
  notes: z.string().optional(),
  language: z.enum(['en', 'no', 'sv', 'da', 'is', 'de']).default('en'),
});

export async function invoiceManager(input: unknown, company_id: string) {
  const parsed = InvoiceSchema.safeParse(input);
  if (!parsed.success) throw new Error(`VALIDATION_ERROR: ${JSON.stringify(parsed.error.flatten())}`);

  const { client_name, items, currency, due_days, notes, language } = parsed.data;

  // Calculate totals
  const subtotal = items.reduce((sum, item) => sum + item.quantity * item.unit_price, 0);
  const vatTotal = items.reduce((sum, item) => sum + item.quantity * item.unit_price * item.vat_rate, 0);
  const total = subtotal + vatTotal;

  const today = new Date();
  const dueDate = new Date(today.getTime() + due_days * 24 * 60 * 60 * 1000);
  const invoiceNumber = `INV-${Date.now().toString(36).toUpperCase()}`;

  // Generate professional invoice text via LLM
  const llm = new LLMClient({ model: 'kimi-k2.5', maxTokens: 1500 });
  const prompt = `Generate a professional invoice email/cover letter in language code "${language}" for:
  - Invoice #: ${invoiceNumber}
  - Client: ${client_name}
  - Subtotal: ${currency} ${subtotal.toFixed(2)}
  - VAT: ${currency} ${vatTotal.toFixed(2)}
  - Total: ${currency} ${total.toFixed(2)}
  - Due date: ${dueDate.toLocaleDateString()}
  ${notes ? `- Notes: ${notes}` : ''}
  Keep it concise, professional, and legally compliant for EU B2B invoicing.`;

  const response = await llm.complete(prompt);
  const cost_eur = CostTracker.estimate('kimi-k2.5', prompt.length + response.length);

  // Save invoice record
  await db.query(
    `INSERT INTO invoices (company_id, invoice_number, client_name, subtotal, vat_total,
     total, currency, due_date, status, line_items, cover_letter, created_at)
     VALUES ($1,$2,$3,$4,$5,$6,$7,$8,'draft',$9,$10,NOW())`,
    [company_id, invoiceNumber, client_name, subtotal, vatTotal, total,
     currency, dueDate.toISOString(), JSON.stringify(items), response.text]
  );

  logger.info({ company_id, invoiceNumber, total, currency }, 'Invoice created');

  return {
    invoice_number: invoiceNumber,
    client_name,
    subtotal: subtotal.toFixed(2),
    vat_total: vatTotal.toFixed(2),
    total: total.toFixed(2),
    currency,
    due_date: dueDate.toISOString().split('T')[0],
    cover_letter: response.text,
    cost_eur,
  };
}

src/skills/finance/fundraising-prep.tsimport { z } from 'zod';
import { LLMClient } from '../../llm/llm-client';
import { CostTracker } from '../../core/cost-tracker';

const FundraisingSchema = z.object({
  company_name: z.string(),
  industry: z.string(),
  stage: z.enum(['pre-seed', 'seed', 'series-a', 'series-b', 'bridge']),
  asking_amount: z.number().positive(),
  currency: z.enum(['EUR', 'GBP', 'USD']).default('EUR'),
  use_of_funds: z.string(),
  current_mrr: z.number().optional(),
  team_size: z.number(),
  traction_metrics: z.string().optional(),
  target_investors: z.enum(['angels', 'vcs', 'family-offices', 'corporate']).default('angels'),
});

export async function fundraisingPrep(input: unknown) {
  const parsed = FundraisingSchema.safeParse(input);
  if (!parsed.success) throw new Error(`VALIDATION_ERROR`);

  const data = parsed.data;
  const llm = new LLMClient({ model: 'kimi-k2.5', maxTokens: 3000 });

  const prompt = `You are a fundraising advisor. Generate a complete fundraising preparation package for:
  Company: ${data.company_name} | Stage: ${data.stage} | Ask: ${data.currency} ${data.asking_amount.toLocaleString()}
  Industry: ${data.industry} | Team: ${data.team_size} people
  Use of funds: ${data.use_of_funds}
  ${data.current_mrr ? `Current MRR: ${data.currency} ${data.current_mrr}` : ''}
  ${data.traction_metrics ? `Traction: ${data.traction_metrics}` : ''}
  Target: ${data.target_investors}

  Produce:
  1. One-paragraph executive summary (investor-ready)
  2. 5-slide pitch deck outline with key points per slide
  3. 3 key financial projections to prepare
  4. 5 most likely investor questions with suggested answers
  5. Suggested valuation range with justification
  Keep all figures and advice specific to the European ${data.stage} ecosystem in 2026.`;

  const response = await llm.complete(prompt);
  const cost_eur = CostTracker.estimate('kimi-k2.5', prompt.length + response.length);

  return {
    package: response.text,
    stage: data.stage,
    ask: `${data.currency} ${data.asking_amount.toLocaleString()}`,
    cost_eur,
    model_used: 'kimi-k2.5',
  };
}

export const a2aServer = Fastify({ logger: false });

// Receive task from peer node
a2aServer.post('/a2a/receive', async (req, reply) => {
  const rawBody = JSON.stringify(req.body);
  const envelope = EnvelopeSchema.safeParse(req.body);
  if (!envelope.success) return reply.status(422).send({ error: 'INVALID_ENVELOPE' });

  const { task_id, from_node, skill_id, payload, cost_gate_eur, signature } = envelope.data;

  // Verify HMAC
  if (!verifyA2ASignature(rawBody, signature, process.env.A2A_SECRET!)) {
    logger.warn({ from_node }, 'A2A signature verification failed');
    return reply.status(401).send({ error: 'INVALID_SIGNATURE' });
  }

  // Cost gate check
  const estimatedCost = CostTracker.estimateSkill(skill_id, payload);
  if (estimatedCost > cost_gate_eur) {
    return reply.status(402).send({ error: 'COST_GATE', estimated_cost: estimatedCost });
  }

  logger.info({ task_id, from_node, skill_id }, 'A2A task received');
  return reply.status(202).send({ task_id, status: 'queued', estimated_cost: estimatedCost });
});

S4-SEA-01 GDPR Erasure Endpoint

Implements right-to-erasure with cascading deletes and audit logging.

src/gdpr/erasure.tsexport async function gdprErasure(req: FastifyRequest, reply: FastifyReply) {
  const { company_id } = req.params as { company_id: string };
  const audit: ErasureAudit = {
    company_id,
    requested_at: new Date().toISOString(),
    tables_cleared: [],
    status: 'pending',
  };

  try {
    // 1. PostgreSQL — cascade delete
    await db.query('DELETE FROM task_logs WHERE company_id = $1', [company_id]);
    audit.tables_cleared.push('task_logs');
    
    // Anonymise company record
    await db.query(
      `UPDATE companies SET name = '[DELETED]', email = $1, settings = '{}',
       stripe_customer_id = NULL, deleted_at = NOW() WHERE id = $2`,
      [`deleted-${company_id.slice(0, 8)}@gdpr.iventure.studio`, company_id]
    );

    // 2. DynamoDB
    for (const table of ['user_entitlements_cache', 'vmoa_agent_state']) {
      await dynamodb.send(new DeleteCommand({ TableName: table, Key: { company_id } }));
    }

    // 3. Redis
    const keys = await redis.keys(`*:${company_id}:*`);
    if (keys.length > 0) await redis.del(...keys);

    // Log audit
    await db.query(
      'INSERT INTO gdpr_erasure_log (company_id, audit_json, created_at) VALUES ($1, $2, NOW())',
      [company_id, JSON.stringify(audit)]
    );

    return reply.status(200).send({ status: 'erased', completed_at: new Date().toISOString() });
  } catch (err) {
    return reply.status(500).send({ error: 'ERASURE_FAILED' });
  }
}

src/security/slack-alerts.tsexport async function sendSecurityAlert(alert: Omit<SlackAlert, 'timestamp'>) {
  const payload = {
    attachments: [{
      color: SEVERITY_COLOR[alert.severity],
      blocks: [
        {
          type: 'section',
          text: {
            type: 'mrkdwn',
            text: `${SEVERITY_EMOJI[alert.severity]} *[${alert.type}]*`
          }
        },
        {
          type: 'section',
          fields: Object.entries(alert.details).map(([key, value]) => ({
            type: 'mrkdwn',
            text: `*${key}:*\n${JSON.stringify(value)}`
          }))
        }
      ]
    }]
  };

  const res = await fetch(process.env.SLACK_SECURITY_WEBHOOK_URL!, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(payload),
  });
}
