S3-BDA-01 Stripe Checkout Integration

Complete TypeScript implementation for subscription checkout sessions.

src/billing/stripe-checkout.tsimport Stripe from 'stripe';
import { FastifyRequest, FastifyReply } from 'fastify';
import { z } from 'zod';
import { db } from '../config/database';
import { logger } from '../observability/logger';

const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!, { apiVersion: '2025-10-01' });

const PRICE_IDS = {
  spark:  process.env.STRIPE_PRICE_SPARK!,
  growth: process.env.STRIPE_PRICE_GROWTH!,
  scale:  process.env.STRIPE_PRICE_SCALE!,
  empire: process.env.STRIPE_PRICE_EMPIRE!,
};

const CheckoutSchema = z.object({
  company_id: z.string().uuid(),
  tier: z.enum(['spark', 'growth', 'scale', 'empire']),
  success_url: z.string().url().optional(),
  cancel_url: z.string().url().optional(),
});

export async function createCheckoutSession(req: FastifyRequest, reply: FastifyReply) {
  const body = CheckoutSchema.safeParse(req.body);
  if (!body.success) return reply.status(422).send({ error: 'VALIDATION_ERROR' });

  const { company_id, tier, success_url, cancel_url } = body.data;

  // Check company exists
  const company = await db.query(
    'SELECT id, name, email, stripe_customer_id FROM companies WHERE id = $1',
    [company_id]
  );
  if (!company.rows[0]) return reply.status(404).send({ error: 'COMPANY_NOT_FOUND' });

  const { name, email, stripe_customer_id } = company.rows[0];

  // Create or get Stripe Customer
  let customerId = stripe_customer_id;
  if (!customerId) {
    const customer = await stripe.customers.create({ email, name, metadata: { company_id } });
    customerId = customer.id;
    await db.query('UPDATE companies SET stripe_customer_id = $1 WHERE id = $2', [customerId, company_id]);
  }

  // Create Session
  const session = await stripe.checkout.sessions.create({
    customer: customerId,
    payment_method_types: ['card'],
    line_items: [{ price: PRICE_IDS[tier], quantity: 1 }],
    mode: 'subscription',
    success_url: success_url ?? `${process.env.APP_URL}/dashboard?checkout=success&tier=${tier}`,
    cancel_url: cancel_url ?? `${process.env.APP_URL}/pricing?checkout=cancelled`,
    metadata: { company_id, tier },
    subscription_data: {
      metadata: { company_id, tier },
      trial_settings: { end_behavior: { missing_payment_method: 'cancel' } },
    },
    allow_promotion_codes: true,
  });

  logger.info({ company_id, tier }, 'Stripe checkout session created');
  return reply.status(200).send({ checkout_url: session.url, session_id: session.id });
}
S3-BDA-02 Stripe Webhook Handler

Handles subscription lifecycle events and entitlement sync.

src/billing/stripe-webhook.tsexport async function stripeWebhookHandler(req: FastifyRequest, reply: FastifyReply) {
  const sig = req.headers['stripe-signature'] as string;
  let event: Stripe.Event;

  try {
    event = stripe.webhooks.constructEvent(req.rawBody as Buffer, sig, process.env.STRIPE_WEBHOOK_SECRET!);
  } catch (err) {
    return reply.status(400).send({ error: 'INVALID_SIGNATURE' });
  }

  switch (event.type) {
    case 'customer.subscription.created':
    case 'customer.subscription.updated':
      await handleSubscriptionUpdate(event.data.object as Stripe.Subscription);
      break;
    case 'customer.subscription.deleted':
      await handleSubscriptionDeleted(event.data.object as Stripe.Subscription);
      break;
    case 'invoice.payment_failed':
      await handlePaymentFailed(event.data.object as Stripe.Invoice);
      break;
  }
  return reply.status(200).send({ received: true });
}

async function handleSubscriptionUpdate(sub: Stripe.Subscription) {
  const company_id = sub.metadata.company_id;
  const tier = TIER_MAP[sub.items.data[0].price.id] ?? 'unknown';
  const status = sub.status;

  await db.query(
    `UPDATE companies SET subscription_tier = $1, subscription_status = $2,
     stripe_subscription_id = $3, updated_at = NOW() WHERE id = $4`,
    [tier, status, sub.id, company_id]
  );

  // Sync to DynamoDB for low-latency checks
  await dynamodb.send(new UpdateCommand({
    TableName: 'user_entitlements_cache',
    Key: { company_id },
    UpdateExpression: 'SET subscription_tier = :t, subscription_status = :s',
    ExpressionAttributeValues: { ':t': tier, ':s': status },
  }));
}
S3-BDA-03 Entitlement Middleware

Enforces subscription limits on every API call.

src/api/middleware/entitlements.tsexport async function entitlementMiddleware(req: FastifyRequest, reply: FastifyReply) {
  const company_id = req.user?.company_id;
  
  // Check DynamoDB Cache
  const result = await dynamodb.send(new GetCommand({
    TableName: 'user_entitlements_cache',
    Key: { company_id },
  }));
  const entitlement = result.Item;

  if (!entitlement) return reply.status(402).send({ error: 'NO_SUBSCRIPTION' });

  if (entitlement.subscription_tier === 'trial') {
    if (new Date(entitlement.trial_ends_at) < new Date()) {
       return reply.status(402).send({ error: 'TRIAL_EXPIRED' });
    }
    const requestedSkill = (req.body as any)?.skill_id;
    if (requestedSkill && !FREE_TRIAL_SKILLS.includes(requestedSkill)) {
       return reply.status(402).send({ error: 'SKILL_NOT_IN_TRIAL' });
    }
  }
  req.entitlement = entitlement;
}

S3-SEA-01 ADR-006: GDPR Architecture

ADR-006: GDPR Data Architecture

Date: 2026-03-18 | Status: ACCEPTED

Context

iVenture Studio processes EU SMB data. Western-market-only focus makes GDPR the primary framework.

Decision

  1. EU Data Residency: All data in AWS eu-central-1 (Frankfurt). No cross-border transfers.
  2. Right to Erasure: Cascading delete across Postgres, DynamoDB, Redis, S3 via DELETE /api/v1/companies/{id}/gdpr/erasure.
  3. Minimisation: Skills log only metadata (cost, duration). Input content not stored unless opted-in.
  4. Lawful Basis: Contract (Service), Legitimate Interest (Security), Consent (Marketing).
S3-SEA-02 Anomaly Detector

Detects cost spikes and attacks using Redis sliding windows.

src/security/anomaly-detector.tsconst CONFIG = {
  costSpike:   { windowMs: 60_000, multiplier: 3 },
  rateLimit:   { windowMs: 60_000, maxRequests: 50 },
};

export async function checkCostAnomaly(company_id: string, cost_eur: number): Promise<boolean> {
  const key = `anomaly:cost:${company_id}:window`;
  
  const pipeline = redis.pipeline();
  pipeline.lpush(key, cost_eur.toString());
  pipeline.ltrim(key, 0, 99); // Keep last 100 txns
  pipeline.lrange(key, 0, -1);
  const results = await pipeline.exec();
  
  const costs = (results![2][1] as string[]).map(Number);
  if (costs.length < 5) return false;
  
  const avg = costs.slice(1).reduce((a, b) => a + b, 0) / (costs.length - 1);
  const isAnomaly = cost_eur > avg * CONFIG.costSpike.multiplier;
  
  if (isAnomaly) {
    await sendSecurityAlert({ 
        type: 'COST_ANOMALY', 
        details: { cost_eur, avg_cost: avg } 
    });
  }
  
  return isAnomaly;
}
