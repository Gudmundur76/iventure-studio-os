S5-BDA-01 PDF Invoice Export

Full TypeScript using Playwright for PDF generation.

src/skills/finance/pdf-export.tsimport { chromium } from 'playwright';
import { db } from '../../config/database';
import { s3 } from '../../config/storage';
import { PutObjectCommand } from '@aws-sdk/client-s3';
import { getSignedUrl } from '@aws-sdk/s3-request-presigner';
import { logger } from '../../observability/logger';

export async function exportInvoicePDF(invoice_id: string, company_id: string): Promise<{ pdf_url: string; expires_at: string }> {
  // Fetch invoice data
  const result = await db.query(
    'SELECT * FROM invoices WHERE id = $1 AND company_id = $2',
    [invoice_id, company_id]
  );
  const invoice = result.rows[0];
  if (!invoice) throw new Error('INVOICE_NOT_FOUND');

  // Generate HTML template for PDF
  const html = generateInvoiceHTML(invoice);

  // Launch headless browser and render PDF
  const browser = await chromium.launch({ args: ['--no-sandbox', '--disable-setuid-sandbox'] });
  const page = await browser.newPage();
  await page.setContent(html, { waitUntil: 'networkidle' });

  const pdfBuffer = await page.pdf({
    format: 'A4',
    margin: { top: '20mm', bottom: '20mm', left: '15mm', right: '15mm' },
    printBackground: true,
  });
  await browser.close();

  // Upload to S3 eu-central-1
  const key = `invoices/${company_id}/${invoice.invoice_number}.pdf`;
  await s3.send(new PutObjectCommand({
    Bucket: process.env.PDF_BUCKET!,
    Key: key,
    Body: pdfBuffer,
    ContentType: 'application/pdf',
    ServerSideEncryption: 'AES256',
    Metadata: { company_id, invoice_id, invoice_number: invoice.invoice_number },
  }));

  // Generate 24-hour pre-signed download URL
  const cmd = new GetObjectCommand({ Bucket: process.env.PDF_BUCKET!, Key: key });
  const pdf_url = await getSignedUrl(s3, cmd, { expiresIn: 86400 });
  const expires_at = new Date(Date.now() + 86400 * 1000).toISOString();

  await db.query('UPDATE invoices SET pdf_url = $1, pdf_generated_at = NOW() WHERE id = $2', [pdf_url, invoice_id]);

  logger.info({ invoice_id, company_id, key }, 'PDF invoice generated');
  return { pdf_url, expires_at };
}

S5-BDA-03 API Key Management
src/api/api-keys.tsimport { createHash, randomBytes } from 'crypto';
import { db } from '../config/database';
import { redis } from '../config/redis';
import { logger } from '../observability/logger';

const KEY_PREFIX = 'ivk_';
const KEY_RATE_LIMIT = 1000; // requests per hour

export function generateApiKey(): { key: string; hash: string } {
  const raw = `${KEY_PREFIX}${randomBytes(32).toString('hex')}`;
  const hash = createHash('sha256').update(raw).digest('hex');
  return { key: raw, hash };
}

export async function createApiKey(company_id: string, name: string, scopes: string[]): Promise<{ key: string; key_id: string }> {
  const { key, hash } = generateApiKey();

  const result = await db.query(
    `INSERT INTO api_keys (company_id, name, key_hash, scopes, created_at)
     VALUES ($1, $2, $3, $4, NOW()) RETURNING id`,
    [company_id, name, hash, JSON.stringify(scopes)]
  );

  logger.info({ company_id, key_id: result.rows[0].id, name }, 'API key created');
  // Return raw key ONCE — never stored in plain text
  return { key, key_id: result.rows[0].id };
}

export async function validateApiKey(raw_key: string): Promise<{ company_id: string; scopes: string[] } | null> {
  if (!raw_key.startsWith(KEY_PREFIX)) return null;

  const hash = createHash('sha256').update(raw_key).digest('hex');
  const result = await db.query(
    `SELECT company_id, scopes FROM api_keys
     WHERE key_hash = $1 AND revoked_at IS NULL AND (expires_at IS NULL OR expires_at > NOW())`,
    [hash]
  );

  if (!result.rows[0]) return null;

  const { company_id, scopes } = result.rows[0];

  // Rate limit: 1000 req/hr per key
  const rateLimitKey = `apikey:ratelimit:${hash.slice(0, 16)}`;
  const count = await redis.incr(rateLimitKey);
  if (count === 1) await redis.expire(rateLimitKey, 3600);
  if (count > KEY_RATE_LIMIT) return null; // 429 handled by caller

  await db.query('UPDATE api_keys SET last_used_at = NOW(), use_count = use_count + 1 WHERE key_hash = $1', [hash]);

  return { company_id, scopes };
}

S5-SEA-01 API Key Security Hardening
src/api/middleware/api-key-auth.tsexport async function apiKeyAuthMiddleware(req: FastifyRequest, reply: FastifyReply) {
  const authHeader = req.headers.authorization;
  if (!authHeader?.startsWith('Bearer ivk_')) {
    return reply.status(401).send({ error: 'MISSING_API_KEY' });
  }

  const raw_key = authHeader.slice(7);
  
  // Check revocation cache first (fast path)
  const revokedKey = `apikey:revoked:${createHash('sha256').update(raw_key).digest('hex').slice(0, 16)}`;
  if (await redis.exists(revokedKey)) {
    return reply.status(401).send({ error: 'KEY_REVOKED' });
  }

  const identity = await validateApiKey(raw_key);
  
  if (!identity) {
    // Rate limit or invalid check ...
    return reply.status(401).send({ error: 'INVALID_API_KEY' });
  }

  // Check scope for this endpoint
  const requiredScope = ENDPOINT_SCOPES[req.routerPath];
  if (requiredScope && !identity.scopes.includes(requiredScope) && !identity.scopes.includes('*')) {
    return reply.status(403).send({ error: 'INSUFFICIENT_SCOPE', required: requiredScope });
  }

  req.user = { company_id: identity.company_id, auth_method: 'api_key' };
}
