S7-BDA-01 Multi-User Team Access
src/api/teams.tsimport { z } from 'zod';
import { randomBytes } from 'crypto';
import { db } from '../config/database';
import { sendEmail } from '../notifications/email';
import { logger } from '../observability/logger';

export type TeamRole = 'owner' | 'admin' | 'member';

const ROLE_PERMISSIONS: Record<TeamRole, string[]> = {
  owner:  ['*'],
  admin:  ['skills:run', 'invoices:*', 'settings:read', 'team:invite', 'team:read'],
  member: ['skills:run', 'invoices:read', 'settings:read'],
};

const InviteSchema = z.object({
  email: z.string().email(),
  role: z.enum(['admin', 'member']).default('member'),
});

export async function inviteTeamMember(company_id: string, inviter_id: string, input: unknown) {
  const parsed = InviteSchema.safeParse(input);
  if (!parsed.success) throw new Error('VALIDATION_ERROR');
  const { email, role } = parsed.data;

  // Check seat limit by subscription tier
  const company = await db.query(
    'SELECT subscription_tier FROM companies WHERE id = $1', [company_id]
  );
  const SEAT_LIMITS: Record<string, number> = {
    trial: 1, spark: 2, growth: 5, scale: 15, empire: 50,
  };
  const tier = company.rows[0]?.subscription_tier ?? 'trial';
  const currentSeats = await db.query(
    "SELECT COUNT(*) FROM team_members WHERE company_id = $1 AND status = 'active'", [company_id]
  );
  if (parseInt(currentSeats.rows[0].count) >= SEAT_LIMITS[tier]) {
    throw new Error(`SEAT_LIMIT_REACHED: ${tier} allows ${SEAT_LIMITS[tier]} seats`);
  }

  const token = randomBytes(32).toString('hex');
  const expires_at = new Date(Date.now() + 7 * 24 * 60 * 60 * 1000); // 7 days

  await db.query(
    `INSERT INTO team_invites (company_id, email, role, token, expires_at, invited_by)
     VALUES ($1,$2,$3,$4,$5,$6)
     ON CONFLICT (company_id, email) DO UPDATE
     SET token = $4, expires_at = $5, role = $3`,
    [company_id, email, role, token, expires_at.toISOString(), inviter_id]
  );

  await sendEmail({
    to: email,
    template: 'team-invite',
    dynamic_data: {
      invite_url: `${process.env.APP_URL}/accept-invite?token=${token}`,
      company_name: company.rows[0]?.name,
      role,
      expires_at: expires_at.toLocaleDateString(),
    },
  });

  logger.info({ company_id, email, role }, 'Team invite sent');
  return { status: 'invited', email, role, expires_at };
}

export async function acceptInvite(token: string, user_id: string) {
  const invite = await db.query(
    `SELECT * FROM team_invites WHERE token = $1 AND expires_at > NOW() AND accepted_at IS NULL`,
    [token]
  );
  if (!invite.rows[0]) throw new Error('INVALID_OR_EXPIRED_INVITE');

  const { company_id, email, role } = invite.rows[0];

  await db.query(
    `INSERT INTO team_members (company_id, user_id, role, joined_at)
     VALUES ($1,$2,$3,NOW())
     ON CONFLICT (company_id, user_id) DO UPDATE SET role = $3`,
    [company_id, user_id, role]
  );
  await db.query('UPDATE team_invites SET accepted_at = NOW() WHERE token = $1', [token]);

  logger.info({ company_id, user_id, role }, 'Team invite accepted');
  return { company_id, role };
}

export function hasPermission(role: TeamRole, required: string): boolean {
  const perms = ROLE_PERMISSIONS[role];
  return perms.includes('*') || perms.includes(required) ||
    perms.some(p => p.endsWith(':*') && required.startsWith(p.slice(0, -2)));
}
S7-BDA-02 Fortnox CSV Export
src/integrations/fortnox-export.tsimport { db } from '../config/database';

export async function exportFortnoxCSV(company_id: string, period: { from: string; to: string }): Promise<string> {
  const result = await db.query(
    `SELECT invoice_number, client_name, subtotal, vat_total, total, currency,
            created_at::date AS invoice_date, due_date, status
     FROM invoices
     WHERE company_id = $1 AND created_at BETWEEN $2 AND $3
     ORDER BY created_at`,
    [company_id, period.from, period.to]
  );

  // Fortnox SIE4/CSV format headers
  const headers = [
    'Fakturanummer',   // Invoice number
    'Kundnamn',        // Client name
    'Fakturadatum',    // Invoice date
    'Förfallodatum',   // Due date
    'Nettobelopp',     // Net amount
    'Mombelopp',       // VAT amount
    'Totalbelopp',     // Total amount
    'Valuta',          // Currency
    'Status',          // Status
  ].join(';');

  const rows = result.rows.map(inv => [
    inv.invoice_number,
    `"${inv.client_name.replace(/"/g, '""')}"`,
    inv.invoice_date,
    inv.due_date,
    inv.subtotal.toString().replace('.', ','),   // Swedish decimal separator
    inv.vat_total.toString().replace('.', ','),
    inv.total.toString().replace('.', ','),
    inv.currency,
    inv.status,
  ].join(';'));

  return [headers, ...rows].join('\r\n'); // Windows line endings for Fortnox
}

src/integrations/slack-bot.tsimport { createHmac, timingSafeEqual } from 'crypto';
import { FastifyRequest, FastifyReply } from 'fastify';
import { db } from '../config/database';
import { runSkill } from '../core/task-executor';
import { logger } from '../observability/logger';

// Slack slash command: /iventure run [skill] [params...]
// Examples:
//   /iventure run competitive-intel Airbnb
//   /iventure run translator "Hello world" --to=is
//   /iventure status
//   /iventure help

function verifySlackSignature(req: FastifyRequest): boolean {
  const timestamp = req.headers['x-slack-request-timestamp'] as string;
  const slackSig  = req.headers['x-slack-signature'] as string;
  if (!timestamp || !slackSig) return false;

  // Reject requests older than 5 minutes
  if (Math.abs(Date.now() / 1000 - parseInt(timestamp)) > 300) return false;

  const sigBase = `v0:${timestamp}:${req.rawBody}`;
  const myHash  = 'v0=' + createHmac('sha256', process.env.SLACK_SIGNING_SECRET!)
    .update(sigBase).digest('hex');

  return timingSafeEqual(Buffer.from(myHash), Buffer.from(slackSig));
}

export async function slackCommandHandler(req: FastifyRequest, reply: FastifyReply) {
  if (!verifySlackSignature(req)) return reply.status(401).send('Invalid signature');

  const { command, text, user_id, team_id } = req.body as any;

  // Resolve company_id from Slack team_id
  const mapping = await db.query(
    'SELECT company_id FROM slack_integrations WHERE slack_team_id = $1', [team_id]
  );
  if (!mapping.rows[0]) {
    return reply.send({
      response_type: 'ephemeral',
      text: '⚠️ iVenture not connected. Visit your dashboard to connect Slack.',
    });
  }
  const { company_id } = mapping.rows[0];

  const args = text.trim().split(/\s+/);
  const subcommand = args[0]?.toLowerCase();

  if (subcommand === 'help') {
    return reply.send({
      response_type: 'ephemeral',
      text: [
        '*iVenture Studio — Slack Commands*',
        '`/iventure run [skill] [input]` — Run an AI skill',
        '`/iventure status` — Show platform status',
        '`/iventure skills` — List available skills',
        '',
        '*Examples:*',
        '`/iventure run competitive-intel "Tesla"` — Analyse a competitor',
        '`/iventure run translator "Hello world" --to=is` — Translate to Icelandic',
      ].join('\n'),
    });
  }

  if (subcommand === 'status') {
    return reply.send({
      response_type: 'ephemeral',
      text: '✅ iVenture Studio is running. All systems operational.',
    });
  }

  if (subcommand === 'run' && args[1]) {
    const skill_id = args[1];
    const rawInput = args.slice(2).join(' ');

    // Acknowledge immediately (Slack requires <3s response)
    reply.send({ response_type: 'in_channel', text: `⏳ Running *${skill_id}*...` });

    // Run skill async and post result back via response_url
    const response_url = (req.body as any).response_url;
    runSkill(skill_id, { input: rawInput }, company_id).then(async (result) => {
      await fetch(response_url, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          response_type: 'in_channel',
          text: `✅ *${skill_id}* complete (€${result.cost_eur?.toFixed(4)})\n\`\`\`${JSON.stringify(result, null, 2).slice(0, 2900)}\`\`\``,
        }),
      });
    }).catch(async (err) => {
      await fetch(response_url, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ text: `❌ Skill failed: ${err.message}` }),
      });
    });

    return; // already sent acknowledgement above
  }

  return reply.send({ response_type: 'ephemeral', text: 'Unknown command. Try `/iventure help`.' });
}

S7-FDA-02 Dark Mode Implementation
components/ThemeToggle.tsxexport function ThemeToggle() {
  const [dark, setDark] = useState(false);

  useEffect(() => {
    const saved = localStorage.getItem('theme');
    const prefersDark = window.matchMedia('(prefers-color-scheme: dark)').matches;
    const isDark = saved ? saved === 'dark' : prefersDark;
    setDark(isDark);
    document.documentElement.classList.toggle('dark', isDark);
  }, []);

  const toggle = () => {
    const next = !dark;
    setDark(next);
    localStorage.setItem('theme', next ? 'dark' : 'light');
    document.documentElement.classList.toggle('dark', next);
  };

  return (
    <button onClick={toggle} className="p-2 rounded-lg hover:bg-slate-100 dark:hover:bg-slate-700 transition"
      aria-label="Toggle dark mode">
      {dark ? '☀️' : '🌙'}
    </button>
  );
}
S7-FDA-04 Fortnox Export UI
components/FortnoxExportButton.tsxexport function FortnoxExportButton({ company_id }: { company_id: string }) {
  const [loading, setLoading] = useState(false);
  const [period, setPeriod] = useState<'current-month' | 'last-month' | 'quarter'>('current-month');

  const handleExport = async () => {
    setLoading(true);
    const res = await fetch(`/api/invoices/export/fortnox?period=${period}&company_id=${company_id}`);
    const blob = await res.blob();
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `iventure-fortnox-${period}-${new Date().toISOString().slice(0,10)}.csv`;
    a.click();
    URL.revokeObjectURL(url);
    setLoading(false);
  };

  return (
    <div className="flex items-center gap-2">
      <select value={period} onChange={e => setPeriod(e.target.value as any)}
        className="text-sm border border-slate-200 rounded-md px-2 py-1.5">
        <option value="current-month">This month</option>
        <option value="last-month">Last month</option>
        <option value="quarter">This quarter</option>
      </select>
      <button onClick={handleExport} disabled={loading}
        className="flex items-center gap-2 bg-blue-600 text-white text-sm px-3 py-1.5 rounded-md hover:bg-blue-700">
        {loading ? '⏳' : '📥'} Export to Fortnox
      </button>
    </div>
  );
}

S7-SEA-01 Multi-User RBAC Security
src/api/middleware/rbac.tsexport function requirePermission(permission: string) {
  return async (req: FastifyRequest, reply: FastifyReply) => {
    const { company_id, user_id } = req.user ?? {};
    if (!company_id || !user_id) return reply.status(401).send({ error: 'UNAUTHENTICATED' });

    const membership = await db.query(
      "SELECT role FROM team_members WHERE company_id = $1 AND user_id = $2 AND status = 'active'",
      [company_id, user_id]
    );

    const ownerCheck = await db.query(
      'SELECT id FROM companies WHERE id = $1 AND owner_user_id = $2', [company_id, user_id]
    );
    if (ownerCheck.rows[0]) {
      req.user!.role = 'owner';
      return; // Owner passes all checks
    }

    const role = membership.rows[0]?.role as TeamRole;
    if (!role) return reply.status(403).send({ error: 'NOT_A_TEAM_MEMBER' });

    if (!hasPermission(role, permission)) {
      logger.warn({ company_id, user_id, role, required: permission }, 'Permission denied');
      return reply.status(403).send({ error: 'INSUFFICIENT_PERMISSION', required: permission, your_role: role });
    }

    req.user!.role = role;
  };
}
S7-SEA-03 ADR-009 Multi-Tenant Security Model
ADR-009: Multi-Tenant Security Model
Date: 2026-03-18 | Status: ACCEPTED | Author: SEA-v1

Context: Sprint 7 introduces team members. We need to ensure strict tenant isolation.

Decision:
1. Every DB query MUST include company_id filter (enforced by custom ESLint rule)
2. JWT contains user_id AND company_id — both verified on every request
3. Team members inherit company_id from invitation — cannot self-assign
4. RBAC roles (owner/admin/member) scope within company, never across companies
5. RLS (Row Level Security) enabled on PostgreSQL for all tenant tables
6. Redis keys always prefixed with company_id
7. S3 paths always prefixed with company_id
