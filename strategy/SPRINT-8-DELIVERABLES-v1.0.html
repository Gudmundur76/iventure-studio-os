src/integrations/datev-export.tsimport { db } from '../config/database';

// DATEV Buchungsstapel (booking stack) CSV format
// Used by German accounting software (Lexware, DATEV, Sevdesk)
export async function exportDATEVCSV(company_id: string, period: { from: string; to: string }): Promise<string> {
  const company = await db.query('SELECT name, vat_number FROM companies WHERE id = $1', [company_id]);
  const { name: company_name, vat_number } = company.rows[0];

  const result = await db.query(
    `SELECT invoice_number, client_name, subtotal, vat_total, total, currency,
            created_at::date AS invoice_date, due_date, status
     FROM invoices
     WHERE company_id = $1 AND created_at BETWEEN $2 AND $3
     ORDER BY created_at`,
    [company_id, period.from, period.to]
  );

  // DATEV header block (mandatory for import)
  const header = [
    '"EXTF";700;21;"Buchungsstapel";5;"";;"";"";"";;0;;""',
    `"${new Date(period.from).toLocaleDateString('de-DE')}";"${new Date(period.to).toLocaleDateString('de-DE')}";"${vat_number ?? ''}";"${company_name}";70`,
  ].join('\r\n');

  // DATEV column headers
  const colHeaders = [
    'Umsatz (ohne Soll/Haben-Kz)',  // Amount
    'Soll/Haben-Kennzeichen',         // Debit/Credit indicator
    'WKZ Umsatz',                     // Currency
    'Kurs',                           // Exchange rate
    'Basis-Umsatz',                   // Base amount
    'WKZ Basis-Umsatz',               // Base currency
    'Konto',                          // Account number
    'Gegenkonto (ohne BU-Schlüssel)', // Contra account
    'BU-Schlüssel',                   // Tax key
    'Belegdatum',                     // Document date
    'Belegfeld 1',                    // Document field 1 (invoice number)
    'Buchungstext',                   // Posting text
    'KOST1 - Kostenstelle',           // Cost centre
  ].join(';');

  const rows = result.rows.map(inv => {
    const amount = parseFloat(inv.total).toFixed(2).replace('.', ',');
    const date = new Date(inv.invoice_date).toLocaleDateString('de-DE', { day: '2-digit', month: '2-digit' }).replace('.', '');
    return [
      amount,
      'S',           // Soll = Debit
      inv.currency,
      '',            // No currency conversion
      '',
      '',
      '1400',        // Standard German AR account (Forderungen)
      '8400',        // Standard revenue account (Erlöse)
      '',
      date,
      inv.invoice_number,
      `"${inv.client_name.replace(/"/g, '""')}"`,
      '',
    ].join(';');
  });

  return [header, colHeaders, ...rows].join('\r\n');
}
S8-BDA-02 Trial Conversion Automation
src/billing/trial-automation.tsimport { db } from '../config/database';
import { redis } from '../config/redis';
import { sendEmail } from '../notifications/email';
import { dispatchWebhook } from '../integrations/outbound-webhook';
import { logger } from '../observability/logger';

// Called by a cron job every hour
export async function processTrialConversions(): Promise<void> {
  const trials = await db.query(`
    SELECT c.id, c.email, c.name, c.trial_ends_at,
           EXTRACT(DAY FROM (c.trial_ends_at - NOW())) AS days_left,
           COUNT(sr.id) AS skills_run,
           MAX(sr.created_at) AS last_activity
    FROM companies c
    LEFT JOIN skill_runs sr ON sr.company_id = c.id
    WHERE c.subscription_tier = 'trial'
      AND c.trial_ends_at > NOW()
      AND c.deleted_at IS NULL
    GROUP BY c.id
    ORDER BY c.trial_ends_at ASC
  `);

  for (const trial of trials.rows) {
    const days = Math.floor(trial.days_left);
    const dedupKey = `trial:nudge:${trial.id}:day${days}`;

    // Skip if already sent this nudge
    if (await redis.exists(dedupKey)) continue;

    let action: string | null = null;

    if (days === 7) {
      action = 'trial-day7-reminder';
    } else if (days === 3) {
      action = 'trial-day3-urgent';
    } else if (days === 1) {
      action = 'trial-day1-final';
    } else if (days === 0) {
      action = 'trial-expired';
    }

    if (action) {
      await sendEmail({
        to: trial.email,
        template: action as any,
        dynamic_data: {
          company_name: trial.name,
          days_left: days,
          skills_run: trial.skills_run,
          upgrade_url: `${process.env.APP_URL}/pricing?source=trial_nudge&day=${days}`,
          highlight_skill: trial.skills_run > 0 ? 'Your most-used skill is ready to scale.' : 'Try competitive-intel free today.',
        },
      });

      // Mark nudge as sent (TTL 48 hours to prevent re-send)
      await redis.setex(dedupKey, 172800, '1');
      logger.info({ company_id: trial.id, action, days }, 'Trial nudge sent');
    }
  }
}

S8-BDA-03 Partner Referral Link System
src/api/referrals.tsimport { createHash, randomBytes } from 'crypto';
import { db } from '../config/database';
import { logger } from '../observability/logger';

const REFERRAL_COMMISSION_RATES: Record<string, number> = {
  silver: 0.20,
  gold:   0.25,
  platinum: 0.30,
};

export async function generateReferralCode(partner_company_id: string): Promise<string> {
  // Generate short, memorable code: first 4 chars of company name + 6 random chars
  const company = await db.query('SELECT name, cortex_tier FROM companies WHERE id = $1', [partner_company_id]);
  const prefix = company.rows[0].name.slice(0, 4).toUpperCase().replace(/[^A-Z0-9]/g, 'X');
  const suffix = randomBytes(3).toString('hex').toUpperCase();
  const code = `${prefix}-${suffix}`;

  await db.query(
    `INSERT INTO referral_codes (partner_company_id, code, cortex_tier, commission_rate, created_at)
     VALUES ($1, $2, $3, $4, NOW())
     ON CONFLICT (partner_company_id) DO UPDATE SET code = $2`,
    [partner_company_id, code, company.rows[0].cortex_tier ?? 'silver',
     REFERRAL_COMMISSION_RATES[company.rows[0].cortex_tier ?? 'silver']]
  );

  logger.info({ partner_company_id, code }, 'Referral code generated');
  return code;
}

export async function trackReferralSignup(code: string, new_company_id: string): Promise<void> {
  const referral = await db.query(
    'SELECT partner_company_id, commission_rate FROM referral_codes WHERE code = $1 AND active = true',
    [code]
  );
  if (!referral.rows[0]) return; // Invalid or inactive code

  const { partner_company_id, commission_rate } = referral.rows[0];

  await db.query(
    `INSERT INTO referral_conversions (referral_code, partner_company_id, referred_company_id,
     commission_rate, status, created_at)
     VALUES ($1, $2, $3, $4, 'pending', NOW())`,
    [code, partner_company_id, new_company_id, commission_rate]
  );

  // Update referred company with attribution
  await db.query(
    'UPDATE companies SET referred_by_code = $1, referred_by_partner = $2 WHERE id = $3',
    [code, partner_company_id, new_company_id]
  );

  logger.info({ code, partner_company_id, new_company_id }, 'Referral signup tracked');
}

export async function calculateMonthlyCommissions(month: string): Promise<Array<{partner: string; amount: number; clients: number}>> {
  const result = await db.query(`
    SELECT rc.partner_company_id, c.name AS partner_name,
           COUNT(rc.id) AS client_count,
           SUM(sub_mrr.mrr * rc.commission_rate) AS commission_eur
    FROM referral_conversions rc
    JOIN companies c ON c.id = rc.partner_company_id
    JOIN (
      SELECT company_id,
        CASE subscription_tier
          WHEN 'spark'  THEN 49
          WHEN 'growth' THEN 149
          WHEN 'scale'  THEN 299
          WHEN 'empire' THEN 999
          ELSE 0
        END AS mrr
      FROM companies WHERE subscription_status = 'active'
    ) sub_mrr ON sub_mrr.company_id = rc.referred_company_id
    WHERE rc.status = 'active' AND TO_CHAR(NOW(), 'YYYY-MM') = $1
    GROUP BY rc.partner_company_id, c.name
    ORDER BY commission_eur DESC
  `, [month]);

  return result.rows.map(r => ({
    partner: r.partner_name,
    amount: parseFloat(r.commission_eur),
    clients: parseInt(r.client_count),
  }));
}
S8-BDA-04 BankID Norway/Sweden Integration
src/auth/bankid.tsimport { FastifyRequest, FastifyReply } from 'fastify';
import { db } from '../config/database';
import { logger } from '../observability/logger';
import { z } from 'zod';

// BankID supports: NO (Vipps BankID), SE (BankID Sweden)
// Used for: enterprise identity verification (not replacing email/password, but adding strong KYC)
// Required for: Norwegian government contracts, Swedish SCALE+ customers

const BANKID_PROVIDERS = {
  NO: { baseUrl: process.env.BANKID_NO_URL!, clientId: process.env.BANKID_NO_CLIENT_ID! },
  SE: { baseUrl: process.env.BANKID_SE_URL!, clientId: process.env.BANKID_SE_CLIENT_ID! },
};

const InitSchema = z.object({
  company_id: z.string().uuid(),
  country: z.enum(['NO', 'SE']),
  personal_number: z.string().optional(), // Sweden requires this
  phone_number: z.string().optional(),    // Norway uses phone
});

export async function initBankIDAuth(req: FastifyRequest, reply: FastifyReply) {
  const parsed = InitSchema.safeParse(req.body);
  if (!parsed.success) return reply.status(422).send({ error: 'VALIDATION_ERROR' });

  const { company_id, country, personal_number, phone_number } = parsed.data;
  const provider = BANKID_PROVIDERS[country];

  // Start auth session with BankID provider
  const response = await fetch(`${provider.baseUrl}/auth`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${process.env[`BANKID_${country}_ACCESS_TOKEN`]}`,
    },
    body: JSON.stringify({
      endUserIp: req.ip,
      ...(personal_number && { personalNumber: personal_number }),
      ...(phone_number && { phoneNumber: phone_number }),
    }),
  });

  const { orderRef, autoStartToken, qrStartToken } = await response.json();

  // Store session for polling
  await db.query(
    `INSERT INTO bankid_sessions (company_id, country, order_ref, status, created_at)
     VALUES ($1, $2, $3, 'pending', NOW())`,
    [company_id, country, orderRef]
  );

  logger.info({ company_id, country, orderRef }, 'BankID auth initiated');

  return reply.send({
    order_ref: orderRef,
    auto_start_token: autoStartToken,
    qr_start_token: qrStartToken,
    poll_url: `/api/v1/auth/bankid/poll/${orderRef}`,
    // Deep link for mobile app
    launch_url: country === 'SE'
      ? `bankid:///?autostarttoken=${autoStartToken}&redirect=null`
      : `vipps:///?token=${autoStartToken}`,
  });
}

export async function pollBankIDAuth(req: FastifyRequest, reply: FastifyReply) {
  const { order_ref } = req.params as { order_ref: string };
  const session = await db.query(
    "SELECT * FROM bankid_sessions WHERE order_ref = $1 AND status = 'pending'", [order_ref]
  );
  if (!session.rows[0]) return reply.status(404).send({ error: 'SESSION_NOT_FOUND' });

  const { company_id, country } = session.rows[0];
  const provider = BANKID_PROVIDERS[country as 'NO' | 'SE'];

  const response = await fetch(`${provider.baseUrl}/collect`, {
    method: 'POST',
    headers: { 'Authorization': `Bearer ${process.env[`BANKID_${country}_ACCESS_TOKEN`]}` },
    body: JSON.stringify({ orderRef: order_ref }),
  });

  const result = await response.json();

  if (result.status === 'complete') {
    const { name, personalNumber } = result.completionData.user;
    await db.query(
      `UPDATE bankid_sessions SET status = 'complete', verified_name = $1, completed_at = NOW()
       WHERE order_ref = $2`,
      [name, order_ref]
    );
    await db.query(
      `UPDATE companies SET bankid_verified = true, bankid_name = $1, bankid_country = $2,
       bankid_verified_at = NOW() WHERE id = $3`,
      [name, country, company_id]
    );
    logger.info({ company_id, country, name }, 'BankID verification complete');
  }

  return reply.send({ status: result.status, hint_code: result.hintCode });
}
